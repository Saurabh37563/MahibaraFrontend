import { useEffect, useState, useRef, useCallback } from "react";
import {
  SSEClient,
  SSEClientOptions,
  SSEEvent,
  SSEEventType,
} from "@/lib/sse-client";

type UseSSEOptions = Omit<SSEClientOptions, "url"> & {
  enabled?: boolean;
};

type EventTypeMap<T extends SSEEventType | "*"> = T extends "*"
  ? SSEEvent
  : Extract<SSEEvent, { type: T }>;

export function useSSE<T extends SSEEventType | "*" = "*">(
  url: string,
  options: UseSSEOptions = {},
) {
  const { enabled = true, token, ...restOptions } = options;
  const clientRef = useRef<SSEClient | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [lastEvent, setLastEvent] = useState<EventTypeMap<T> | null>(null);
  const isEnabledRef = useRef(enabled);

  // Store the options in a ref to avoid recreating the client on every option change
  const optionsRef = useRef(restOptions);
  optionsRef.current = restOptions;
  isEnabledRef.current = enabled;

  // Memoize the client initialization with stable dependencies
  const initializeClient = useCallback(() => {
    if (!clientRef.current) {
      clientRef.current = new SSEClient({
        url,
        token,
        onConnect: () => {
          if (isEnabledRef.current) {
            setIsConnected(true);
          }
        },
        onDisconnect: (reason) => {
          setIsConnected(false);
          console.log(`[SSE] Disconnected: ${reason}`);
        },
        onError: (err) => {
          setError(err);
          console.error("[SSE] Error:", err);
        },
        ...optionsRef.current,
      });
    }
  }, [url, token]); // Only depend on url and token, not restOptions

  // Single effect to handle connection lifecycle
  useEffect(() => {
    if (!enabled) {
      // Disconnect if disabled
      if (clientRef.current && isConnected) {
        console.log("[SSE] Disabling connection due to enabled=false");
        clientRef.current.disconnect();
      }
      return;
    }

    // Initialize client if needed
    if (!clientRef.current) {
      initializeClient();
    }

    // Connect if not already connected
    if (clientRef.current && !isConnected) {
      console.log("[SSE] Connecting...");
      clientRef.current.connect();
    }

    return () => {
      if (clientRef.current && isConnected) {
        console.log("[SSE] Cleanup - disconnecting");
        clientRef.current.disconnect();
      }
    };
  }, [enabled, initializeClient, isConnected]); // Include isConnected for proper cleanup

  const addEventListener = useCallback(
    <E extends SSEEventType | "*">(
      eventType: E,
      handler: (event: EventTypeMap<E>) => void,
    ): (() => void) => {
      if (!clientRef.current) {
        initializeClient();
      }
      return (
        clientRef.current?.on(
          eventType,
          handler as (event: SSEEvent) => void,
        ) ?? (() => undefined)
      );
    },
    [initializeClient],
  );

  const connect = useCallback(() => {
    if (clientRef.current && !isConnected) {
      clientRef.current.connect();
    }
  }, [isConnected]);

  const disconnect = useCallback(() => {
    if (clientRef.current) {
      clientRef.current.disconnect();
    }
  }, []);

  return {
    isConnected,
    lastEvent,
    error,
    addEventListener,
    connect,
    disconnect,
  } as const;
}
